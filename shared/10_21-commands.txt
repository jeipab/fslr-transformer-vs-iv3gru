===============================================================================
FSLR PREPROCESSING PIPELINE - GREETINGS-ONLY DATASET
===============================================================================

Step 1: Preprocessing (Extract Keypoints + IV3 Features)
-------------------------------------------------------------------------------
python -m preprocessing.core.preprocess \
    data/raw/Greetings-Only \
    data/processed \
    --write-keypoints \
    --write-iv3-features \
    --workers 4 \
    --disable-parquet \
    --occ-detailed \
    --batch-size 64

Output: data/processed/Greetings-Only/*.npz (1217 files)


Step 2: Labels Generation (Create labels.csv from NPZ metadata)
-------------------------------------------------------------------------------
python data/splitting/assign.py \
    --directory data/processed/Greetings-Only \
    --reference data/labels_reference.csv

Output: data/processed/Greetings-Only/labels.csv


Step 3: Data Splitting (80/20 Train/Val Split per Signer)
-------------------------------------------------------------------------------
python data/splitting/data_split.py \
    --processed-root data/processed/Greetings-Only \
    --labels data/processed/Greetings-Only/labels.csv \
    --out-root data/processed \
    --copy \
    --train-ratio 0.8

Output: 
  - data/processed/keypoints_train/ (~973 files)
  - data/processed/keypoints_val/ (~244 files)
  - data/processed/train_labels.csv
  - data/processed/val_labels.csv
